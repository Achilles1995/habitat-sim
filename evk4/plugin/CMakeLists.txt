# Alternative CMake build (e.g. Windows / building against a MuJoCo release archive).
#   cmake -S plugin -B build -DMUJOCO_DIR=/path/to/mujoco   (dir containing include/ and lib/)
cmake_minimum_required(VERSION 3.16)
project(evk4_event_camera CXX)
set(CMAKE_CXX_STANDARD 17)
set(MUJOCO_DIR "" CACHE PATH "MuJoCo install/package dir (has include/mujoco)")
find_library(MUJOCO_LIB NAMES mujoco PATHS ${MUJOCO_DIR} ${MUJOCO_DIR}/lib ${MUJOCO_DIR}/bin NO_DEFAULT_PATH)
add_library(evk4_event_camera SHARED evk4_event_camera.cc)
target_include_directories(evk4_event_camera PRIVATE ${MUJOCO_DIR}/include)
find_package(Threads REQUIRED)
target_link_libraries(evk4_event_camera PRIVATE ${MUJOCO_LIB} Threads::Threads)
if(NOT MSVC)
  target_compile_options(evk4_event_camera PRIVATE -O3)
endif()
